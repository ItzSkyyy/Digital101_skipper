How to use
  1. Show your bookmarks bar — Ctrl+Shift+B on Windows  /  ⌘+Shift+B on Mac
  2. Drag both buttons above into the bookmarks bar
  3. Go to your FutureSkills Prime pathway page — the one listing all your modules
  4. Click FSP Auto-Complete — it loops through all modules, skipping quizzes, and tells you how many quizzes remain when done
  5. For each quiz, open the quiz tab, click Start Quiz, wait for the first question to appear, then click FSP Quiz Auto-Answer — it reads the correct answers from the review page and submits the form directly
  6. After the quiz submits, click Process My Result & Close to finalise it
  7. Already-completed modules are detected and skipped automatically
  8. An alert shows Completed and Skipped counts when the module loop finishes

— Script 1: Pathway page —

Open F12 → Console on your FSP pathway page, paste and press Enter.

(async function fspAutoComplete() {
  const ANIM_WAIT  = 1400;
  const ARM_WAIT   = 700;
  const AFTER_DONE = 1000;
  const wait = ms => new Promise(r => setTimeout(r, ms));

  function getContentID(vc) {
    const elId = vc.id || '';
    if (elId.startsWith('view_content_')) return elId.replace('view_content_', '');
    const oc = vc.getAttribute('onclick') || '';
    const s = oc.indexOf('('), e = oc.indexOf(',');
    if (s > -1 && e > -1) return oc.slice(s + 1, e).replace(/[^0-9]/g, '');
    return '';
  }

  function getMarkID(mc) {
    const oc = mc?.getAttribute('onclick') || '';
    const s = oc.indexOf('('), e = oc.indexOf(')');
    if (s > -1 && e > -1) return oc.slice(s + 1, e).replace(/[^0-9]/g, '');
    return '';
  }

  const allGS = Array.from(document.querySelectorAll('button.getStarted'));
  console.log(`🚀 Found ${allGS.length} module(s)`);
  let done = 0, skipped = 0, quizCount = 0;

  for (let i = 0; i < allGS.length; i++) {
    const gs = allGS[i];
    const ctx = (gs.closest('div,li,section') || document.body).textContent.toLowerCase();
    if (ctx.includes('quiz') || ctx.includes('assessment')) {
      console.log(`⏭ Module ${i + 1}: Quiz — use Script 2 in that tab`);
      quizCount++; skipped++; continue;
    }
    console.log(`\n📦 Module ${i + 1} / ${allGS.length}`);
    gs.click();
    await wait(ANIM_WAIT);

    const vc = document.querySelector('a.post_cont_view_content');
    if (!vc) { skipped++; gs.click(); await wait(400); continue; }

    const contentId = getContentID(vc);
    if (!contentId) { skipped++; gs.click(); await wait(400); continue; }

    try { pre_mark_as_complete(contentId, 'false'); }
    catch(e) { console.warn('  ❌', e); skipped++; gs.click(); await wait(400); continue; }

    await wait(ARM_WAIT);

    const mc = document.querySelector('button#markasComplete');
    const markId = getMarkID(mc);
    if (markId) {
      try {
        mark_as_complete_hub_product(parseInt(markId));
        await wait(AFTER_DONE);
        done++;
        console.log('  🎉 Complete!');
      } catch(e) { console.warn('  ❌', e); skipped++; }
    } else {
      console.log('  ⏭ Already done — skipping.'); skipped++;
    }
    gs.click();
    await wait(600);
  }
  console.log(`\n✅ Completed: ${done}  |  ⏭ Skipped: ${skipped}${quizCount ? `  |  🧠 Quizzes: ${quizCount} (run Script 2 in each quiz tab)` : ''}`);
})();

— Script 2: Quiz tab (after clicking Start Quiz) —

Open the quiz tab, click Start Quiz, wait for first question, then paste this.

(async function fspQuizAutoAnswer() {
  // Clicks the correct answer and Submit for each question one by one.
  // Run AFTER clicking Start Quiz, once the first question is visible.

  const obj  = document.getElementById('content_id');
  const odoc = obj && (obj.contentDocument || obj.contentWindow?.document);
  if (!odoc) { alert('No quiz found. Make sure you clicked Start Quiz first.'); return; }

  const wait = ms => new Promise(r => setTimeout(r, ms));

  // Click Start Quiz if present
  const startBtn = [...odoc.querySelectorAll('a,button')].find(el => el.textContent.trim().toLowerCase() === 'start quiz');
  if (startBtn) { startBtn.click(); await wait(1200); }

  // Correct answers are in the page scripts as "is_true"=>1
  const scripts = [...odoc.querySelectorAll('script')].map(s => s.textContent).join('');
  const q = '"'; const gt = '=>';
  function isCorrect(v) {
    return scripts.includes('answer_id'+q+gt+v+', '+q+'answer_hint'+q+gt+q+q+', '+q+'is_true'+q+gt+'1}')
        || scripts.includes('answer_id'+q+gt+v+','+q+'answer_hint'+q+gt+q+q+','+q+'is_true'+q+gt+'1}');
  }

  function selectRadio(r) {
    r.checked = true;
    r.dispatchEvent(new Event('change', {bubbles: true}));
    r.dispatchEvent(new MouseEvent('click', {bubbles: true}));
    const labels = odoc.querySelectorAll('label');
    for (const l of labels) { if (l.htmlFor === r.id) { l.click(); break; } }
    const cl = r.closest('label'); if (cl) cl.click();
  }

  function freshDoc() {
    const o = document.getElementById('content_id');
    return o && (o.contentDocument || (o.contentWindow && o.contentWindow.document));
  }

  function getVisible() {
    for (const q of odoc.querySelectorAll('[id^="quiz_question"]')) {
      const r = q.getBoundingClientRect();
      if (r.width > 0 || r.height > 0) return q;
    }
    return odoc.querySelector('[id^="quiz_question"]');
  }

  const total = odoc.querySelectorAll('[id^="quiz_question"]').length;
  console.log(`🧠 Found ${total} question(s)`);
  let answered = 0;

  for (let attempt = 0; attempt < total + 2; attempt++) {
    const qBlock = getVisible();
    if (!qBlock) break;
    const radios = qBlock.querySelectorAll('input[type="radio"]');
    if (!radios.length) break;
    let correct = [...radios].find(r => isCorrect(r.value)) || radios[0];
    console.log(`  Q${attempt+1}: ${correct.value} ${isCorrect(correct.value) ? '✅' : '⚠️ fallback'}`);
    selectRadio(correct);
    await wait(600);
    if (!qBlock.querySelector('input[type="radio"]:checked')) { selectRadio(correct); await wait(400); }
    const submitBtn = qBlock.querySelector('a.submit_quiz_step1');
    if (!submitBtn) break;
    submitBtn.click(); answered++; await wait(900);
    const nextBtn = qBlock.querySelector('button.quizz_btn');
    if (nextBtn) { nextBtn.click(); await wait(700); }
  }

  // View Result
  await wait(500);
  const viewResult = odoc.querySelector('button.submit_quiz_btn');
  if (viewResult) { viewResult.click(); await wait(1500); }

  // Continue — re-fetch doc in case it reloaded
  odoc = freshDoc();
  const contBtn = [...odoc.querySelectorAll('a,button')].find(el => el.textContent.trim().toLowerCase() === 'continue');
  if (contBtn) { contBtn.click(); await wait(2500); }

  // Process My Result & Close — poll with fresh doc
  for (let t = 0; t < 10; t++) {
    odoc = freshDoc();
    const processBtn = odoc && odoc.querySelector('a.prcoess_result');
    if (processBtn) { processBtn.click(); break; }
    await wait(500);
  }
})();

If timings are off, tweak ANIM_WAIT, ARM_WAIT, or AFTER_DONE in Script 1.
If the quiz script can't find correct answers, it falls back to the first option per question.
