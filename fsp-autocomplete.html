How to use
  1. Show your bookmarks bar — Ctrl+Shift+B on Windows  /  ⌘+Shift+B on Mac
  2. Drag FSP Auto-Complete (orange) to your bookmarks bar
  3. Drag FSP Quiz Auto-Answer (teal) to your bookmarks bar
  4. Go to your FSP pathway page and click FSP Auto-Complete — it completes all modules automatically, enrolling where needed
  5. When a quiz opens in a new tab, switch to it and click FSP Quiz Auto-Answer — it starts, answers, and submits the quiz
  6. Already-completed modules are detected and skipped automatically
Script 1 — pathway page 
(async function() {
  const ARM = 700, AFTER = 1000;
  const wait = ms => new Promise(r => setTimeout(r, ms));

  async function waitFor(sel, maxMs) {
    const start = Date.now();
    while (Date.now() - start < maxMs) {
      const el = document.querySelector(sel);
      if (el) return el;
      await wait(150);
    }
    return null;
  }

  function getID(vc) {
    const id = vc.id || '';
    if (id.startsWith('view_content_')) return id.replace('view_content_', '');
    const oc = vc.getAttribute('onclick') || '';
    const s = oc.indexOf('('), e = oc.indexOf(',');
    if (s > -1 && e > -1) return oc.slice(s + 1, e).replace(/[^0-9]/g, '');
    return '';
  }

  function getMCID(mc) {
    const oc = mc?.getAttribute('onclick') || '';
    const s = oc.indexOf('('), e = oc.indexOf(')');
    if (s > -1 && e > -1) return oc.slice(s + 1, e).replace(/[^0-9]/g, '');
    return '';
  }

  function findEnrolNear(row) {
    for (const b of row.querySelectorAll('a,button')) {
      const t = b.textContent.trim().toLowerCase();
      if (t === 'enrol' || t === 'enroll') return b;
    }
    return null;
  }

  // Collect all module rows
  let allRows = [...document.querySelectorAll('.content_hub_list_item,.pathway_content_item,.hub_product_row,li.hub_row')];
  if (!allRows.length) {
    const seen = new Set();
    document.querySelectorAll('button.getStarted').forEach(b => {
      const r = b.closest('li,tr,div.row,div.item') || b.parentElement.parentElement;
      if (!seen.has(r)) { seen.add(r); allRows.push(r); }
    });
    document.querySelectorAll('a,button').forEach(b => {
      const t = b.textContent.trim().toLowerCase();
      if (t === 'enrol' || t === 'enroll') {
        const r = b.closest('li,tr,div.row,div.item') || b.parentElement.parentElement;
        if (!seen.has(r)) { seen.add(r); allRows.push(r); }
      }
    });
  }

  if (!allRows.length) { alert('No modules found.'); return; }

  let done = 0, skip = 0, quizCount = 0;

  for (let i = 0; i < allRows.length; i++) {
    const row = allRows[i];
    const ctx = row.textContent.toLowerCase();
    if (ctx.includes('quiz') || ctx.includes('assessment')) { quizCount++; skip++; continue; }

    let gs = row.querySelector('button.getStarted');
    if (!gs) {
      const enrolBtn = findEnrolNear(row);
      if (!enrolBtn) { skip++; continue; }
      enrolBtn.click();
      await wait(300);
      gs = await waitFor('button.getStarted', 5000);
      if (!gs) { skip++; continue; }
      await wait(500);
    }

    gs.click();
    const vc = await waitFor('a.post_cont_view_content', 4000);
    if (!vc) { skip++; gs.click(); await wait(400); continue; }

    const cid = getID(vc);
    if (!cid) { skip++; gs.click(); await wait(400); continue; }

    try { pre_mark_as_complete(cid, 'false'); }
    catch(e) { skip++; gs.click(); await wait(400); continue; }

    await wait(ARM);

    let mc = null;
    for (let t = 0; t < 10; t++) {
      mc = document.querySelector('button#markasComplete');
      if (getMCID(mc)) break;
      await wait(200);
    }

    const mcid = getMCID(mc);
    if (mcid) {
      try { mark_as_complete_hub_product(parseInt(mcid)); await wait(AFTER); done++; }
      catch(e) { skip++; }
    } else { skip++; }

    gs.click();
    await wait(600);
  }

  alert(`Done!\n\nCompleted: ${done}\nSkipped: ${skip}${quizCount ? '\n\n' + quizCount + ' quiz(zes) — use FSP Quiz button in quiz tab' : ''}`);
})();
Script 2 — quiz tab 
(async function() {
  const wait = ms => new Promise(r => setTimeout(r, ms));

  function freshDoc() {
    const o = document.getElementById('content_id');
    return o && (o.contentDocument || (o.contentWindow && o.contentWindow.document));
  }
  function findBtn(doc, text) {
    for (const el of doc.querySelectorAll('a,button'))
      if (el.textContent.toLowerCase().includes(text)) return el;
    return null;
  }
  function selectInput(odoc, r) {
    r.checked = true;
    r.dispatchEvent(new Event('change', {bubbles: true}));
    r.dispatchEvent(new MouseEvent('click', {bubbles: true}));
    for (const l of odoc.querySelectorAll('label'))
      if (l.htmlFor === r.id) { l.click(); break; }
    const cl = r.closest('label'); if (cl) cl.click();
  }
  function isCorrect(scripts, q, gt, v) {
    const p1 = 'answer_id'+q+gt+v+', '+q+'answer_hint'+q+gt+q+q+', '+q+'is_true'+q+gt+'1}';
    const p2 = 'answer_id'+q+gt+v+','+q+'answer_hint'+q+gt+q+q+','+q+'is_true'+q+gt+'1}';
    return scripts.includes(p1) || scripts.includes(p2);
  }

  let odoc = freshDoc();
  if (!odoc) { alert('No quiz found. Make sure the quiz tab is open.'); return; }

  const startBtn = findBtn(odoc, 'start quiz');
  if (startBtn) { startBtn.click(); await wait(1200); }

  const scripts = [...odoc.querySelectorAll('script')].map(s => s.textContent).join('');
  const q = '"', gt = '=>';

  function getVisible() {
    for (const qb of odoc.querySelectorAll('[id^=quiz_question]')) {
      const r = qb.getBoundingClientRect();
      if (r.width > 0 || r.height > 0) return qb;
    }
    return null;
  }
  function getQType(qBlock) {
    const btn = qBlock.querySelector('a.submit_quiz_step1');
    if (!btn) return 'Single';
    return (btn.getAttribute('onclick') || '').includes('Multiple') ? 'Multiple' : 'Single';
  }

  const total = odoc.querySelectorAll('[id^=quiz_question]').length;
  for (let attempt = 0; attempt < total; attempt++) {
    const qBlock = getVisible(); if (!qBlock) break;
    const submitBtn = qBlock.querySelector('a.submit_quiz_step1'); if (!submitBtn) break;

    if (getQType(qBlock) === 'Multiple') {
      let any = false;
      for (const cb of qBlock.querySelectorAll('input[type=checkbox]'))
        if (isCorrect(scripts, q, gt, cb.value)) { selectInput(odoc, cb); any = true; await wait(200); }
      if (!any) selectInput(odoc, qBlock.querySelector('input[type=checkbox]'));
    } else {
      const radios = [...qBlock.querySelectorAll('input[type=radio]')];
      const correct = radios.find(r => isCorrect(scripts, q, gt, r.value)) || radios[0];
      selectInput(odoc, correct); await wait(600);
      if (!qBlock.querySelector('input:checked')) { selectInput(odoc, correct); await wait(400); }
    }

    await wait(300);
    submitBtn.click(); await wait(900);
    const nextBtn = qBlock.querySelector('button.quizz_btn');
    if (nextBtn) { nextBtn.click(); await wait(700); }
  }

  await wait(500);
  const viewResult = odoc.querySelector('button.submit_quiz_btn');
  if (viewResult) { viewResult.click(); await wait(1500); }

  odoc = freshDoc();
  const contBtn = findBtn(odoc, 'continue');
  if (contBtn) { contBtn.click(); await wait(2500); }

  for (let t = 0; t < 10; t++) {
    odoc = freshDoc();
    const processBtn = odoc && odoc.querySelector('a.prcoess_result');
    if (processBtn) { processBtn.click(); break; }
    await wait(500);
  }
})();

If timings are off, tweak ARM or AFTER in Script 1.
